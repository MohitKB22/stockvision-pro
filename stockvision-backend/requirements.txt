# --- Data validation / settings ----------------------------------------------
pydantic==2.12.3   # bumped from 2.10.4: its pydantic-core==2.27.2 dependency
                    # has no wheel for Python 3.14, and pip's from-source
                    # fallback fails harder than scipy/scikit-learn/shap did
                    # above -- it's not just a missing wheel, the specific
                    # PyO3 (Rust/Python bridge) version pydantic-core 2.27.2
                    # depends on doesn't support building against Python
                    # 3.14's C API at all, regardless of what compilers are
                    # installed. 2.12.3 is the latest patch of the first
                    # pydantic 2.x minor version whose pydantic-core pin
                    # (2.41.4) has a real 3.14 wheel -- chosen over jumping
                    # straight to the newest release (2.13.x) to minimize
                    # API drift from what this project was built against.
                    # pydantic-settings==2.7.1 only requires pydantic>=2.7.0,
                    # so it did not need a matching bump.
pydantic-settings==2.7.1
# email-validator removed with schemas/auth.py — nothing validates an email any more.

# --- Database / ORM / migrations ----------------------------------------------
sqlalchemy==2.0.51   # bumped from 2.0.36: that version predates several
                      # SQLAlchemy fixes specifically for Python 3.14's PEP
                      # 649 (deferred annotation evaluation, the default
                      # behavior in 3.14 with no `from __future__ import
                      # annotations` needed) -- confirmed via SQLAlchemy's own
                      # changelog, which lists multiple "Fixed issue ... with
                      # Python 3.14's PEP 649 feature" entries in 2.0.x
                      # releases after 2.0.36. Symptom on 2.0.36 + Python
                      # 3.14: declarative mapping crashes with
                      # `TypeError: descriptor '__getitem__' requires a
                      # 'typing.Union' object but received a 'tuple'` on ANY
                      # model using `Mapped[X | None]` (PEP 604 union syntax)
                      # -- i.e. most of this project's nullable columns.
                      # Stayed on the 2.0.x line (not 2.1) to minimize API
                      # drift; re-verified against the full test suite.
alembic==1.14.0
psycopg2-binary==2.9.10         # PostgreSQL driver (production target)

# --- Auth / security -----------------------------------------------------------

# against code nobody is maintaining. Three fewer dependencies is also three
# fewer CVE surfaces to track.
